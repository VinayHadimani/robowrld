     1→     1→/**
     2→     2→ * The component schema — the platform's single source of truth.
     3→     3→ *
     4→     4→ * This file is what turns "a nice 3D viewer" into Pillar 1 of the robotics
     5→     5→ * creation platform. Every library component is described as DATA:
     6→     6→ *
     7→     7→ *   identity    → what it is, who makes it, what it costs (real-world commerce)
     8→     8→ *   specs       → the headline engineering numbers from the datasheet
     9→     9→ *   physics     → mass, materials, electrical load — what a simulation engine
    10→    10→ *                 (Godot later) needs to make the virtual behave like the real
    11→    11→ *   ai          → a machine-readable profile the AI generation layer (Pillar 2)
    12→    12→ *                 reads when a user says "build me a robot that avoids walls"
    13→    13→ *   model       → where the 3D geometry comes from (procedural today, glTF
    14→    14→ *                 from OpenSCAD/STL exports later — the schema accepts both)
    15→    15→ *
    16→    16→ * The geometry itself stays procedural in TypeScript (see ServoMotor.tsx):
    17→    17→ * procedural models can be parameterized, are tiny to download, and render
    18→    18→ * interactively. The DB stores the catalog + ownership; this file is the
    19→    19→ * definition. Everything here is JSON-serializable on purpose so it can be
    20→    20→ * served straight to browsers, the AI layer, and the future build-instruction
    21→    21→ * exporter without transformation.
    22→    22→ */
    23→    23→

    30→    30→
    31→    31→export type ComponentStatus = 'available' | 'soon'
    32→    32→export type ComponentCategory = 'Actuators' | 'Sensors' | 'Power' | 'Controller'
    33→    33→
    34→    34→export interface SpecRow {
    35→    35→  label: string
    36→    36→  value: string
    37→    37→}
    38→    38→
    39→    39→export interface PhysicsProperties {
    40→    40→  /** Mass in grams — the number a physics engine steps with. */
    41→    41→  mass: number
    42→    42→  /** Footprint in mm: length × width × height. */
    43→    43→  dimensions: { l: number; w: number; h: number }
    44→    44→  /** What each part is made of — feeds material physics + cost realism. */
    45→    45→  materials: Array<{ part: string; material: string }>
    46→    46→  /** Electrical load profile. */
    47→    47→  electrical: Array<{ label: string; value: string }>
    48→    48→  /** Mechanical ratings relevant to simulation. */
    49→    49→  mechanical: Array<{ label: string; value: string }>
    50→    50→  /** One-paragraph note on how the sim engine should treat this part. */
    51→    51→  simNotes: string
    52→    52→}
    53→    53→
    54→    54→export interface CommerceInfo {
    55→    55→  manufacturer: string
    56→    56→  partNumber: string
    57→    57→  /** Typical street price range in USD. */
    58→    58→  priceUsd: [number, number]
    59→    59→  /** INR equivalent — the first market this platform targets. */
    60→    60→  priceInr: [number, number]
    61→    61→  availability: string
    62→    62→  /** Where a student can actually buy it. */
    63→    63→  vendors: string[]
    64→    64→}
    65→    65→
    66→    66→export interface AiProfile {
    67→    67→  /** One line the AI layer reads first. */
    68→    68→  summary: string
    69→    69→  /** What this component can do in an assembly. */
    70→    70→  capabilities: string[]
    71→    71→  /** Hard limits the AI must respect when designing. */
    72→    72→  constraints: string[]
    73→    73→  /** Situations where this is the right pick. */
    74→    74→  bestFor: string[]
    75→    75→}
    76→    76→
    77→    77→export type ModelSource =
    78→    78→  | {
    79→    79→      /** Code-generated parametric geometry (an R3F component export). */
    80→    80→      kind: 'procedural'
    81→    81→      /** Name of the exported React Three Fiber component. */
    82→    82→      component: string
    83→    83→      /** True if the mesh matches the spec-sheet dimensions exactly. */
    84→    84→      dimensionAccurate: boolean
    85→    85→      note: string
    86→    86→    }
    87→    87→  | {
    88→    88→      /** External asset (future: OpenSCAD → STL → glTF pipeline). */
    89→    89→      kind: 'gltf'
    90→    90→      url: string
    91→    91→    }

    93→    93→export interface RoboticsComponent {
    94→    94→  id: string
    95→    95→  name: string
    96→    96→  category: ComponentCategory
    97→    97→  status: ComponentStatus
    98→    98→  tagline: string
    99→    99→  /** Long-form description for the component detail page. */
   100→   100→  description: string
   101→   101→  specs: SpecRow[]
   102→   102→  physics: PhysicsProperties
   103→   103→  commerce: CommerceInfo
   104→   104→  ai: AiProfile
   105→   105→  model: ModelSource
   106→   106→  /** Connection contract (ports + DOF) — defined in connections.ts. */
   107→   107→  connectionModel?: ComponentModel
   108→   108→}

   619→   619→export function aiCatalogPayload() {

   621→   621→    schema: 'robotics-platform.catalog/1.0',
