     3→     3→/**
     4→     4→ * PlaygroundStage — the 3D canvas where placed components live.
     5→     5→ *
     6→     6→ * Interaction model:
     7→     7→ *   - pointer-down on a component → grab it (orbit controls freeze); its
     8→     8→ *     whole connected group comes along — wired things move together
     9→     9→ *   - move → the group slides along the ground plane (imperative, no
    10→    10→ *     re-renders); the nearest foreign component in reach gets a pulsing
    11→    11→ *     ring: green when the pair may REALLY connect, red when it may not
    12→    12→ *   - release near a CONNECTABLE neighbour → the piece slides into the
    13→    13→ *     connected spot as a preview and the app ASKS FOR PERMISSION; only a
    14→    14→ *     "Connect" answer wires them together
    15→    15→ *   - release near a NON-connectable pair (e.g. a servo onto a 9V
    16→    16→ *     battery — 9 V is outside its 4.8–7.2 V range) → nothing happens;
    17→    17→ *     they simply can't be connected, like real life
    18→    18→ *   - a click without movement → select (ring + panel)
    19→    19→ *   - Run → only components with a valid wire path to a power source
    20→    20→ *     animate; everything else stays still. Battery and board LEDs light
    21→    21→ *     while they are actually powering something.
    22→    22→ *
    23→    23→ * Wiring is drawn BETWEEN REAL PORTS: each connection renders a cable
    24→    24→ * from the provider's port (the battery's brass +/− snap terminals, the
    25→    25→ * controller's 5 V header) to the consumer's port (a motor's leads, a
    26→    26→ * servo's signal cable, the board's VIN jack). Power splices are a red +
    27→    27→ * black pair (red always on +); servo-style links are a 3-wire ribbon.
    28→    28→ * When one battery feeds several components, their wire pairs fan out
    29→    29→ * side by side from the same two terminals — like stacked spade lugs.
    30→    30→ *
    31→    31→ * Positions are applied ONCE on mount (never via the `position` prop) so a
    32→    32→ * re-render mid-drag can never snap a component back to a stale spot.
    33→    33→ */

   133→   133→/**
   134→   134→ * One cable between two anchor objects. The tube geometry is rebuilt in
   135→   135→ * useFrame only when an endpoint actually moved — so cables stay glued
   136→   136→ * to their components even during imperative (mid-drag) moves, at zero
   137→   137→ * React render cost. `offsetA` shifts the provider-side clip point
   138→   138→ * (battery terminal fan-out); ribbon wires shift perpendicular to the
   139→   139→ * run so the three leads sit side by side like a flat cable.
   140→   140→ */
   141→   141→function Cable({
   142→   142→  anchors,
   143→   143→  parentId,
   144→   144→  parentPortId,
   145→   145→  childId,
   146→   146→  childPortId,
   147→   147→  side,
   148→   148→  color,
   149→   149→  live,
   150→   150→  spread,
   151→   151→  lateral,
   152→   152→  radius,
   153→   153→}: {
   154→   154→  anchors: React.RefObject<AnchorMap>
   155→   155→  parentId: string
   156→   156→  parentPortId: string
   157→   157→  childId: string
   158→   158→  childPortId: string
   159→   159→  /** Which lead of each port this cable carries ('red' | 'black'). */
   160→   160→  side: 'red' | 'black'
   161→   161→  color: string
   162→   162→  live: boolean
   163→   163→  /** Provider-side lateral fan-out at a shared terminal (world X). */
   164→   164→  spread: number
   165→   165→  /** Consumer-side perpendicular offset (ribbon wires; world space). */
   166→   166→  lateral: number
   167→   167→  radius: number
   168→   168→}) {
   169→   169→  const mesh = useRef<THREE.Mesh>(null)
   170→   170→  const lastA = useRef(new THREE.Vector3(1e9, 1e9, 1e9))
   171→   171→  const lastB = useRef(new THREE.Vector3(1e9, 1e9, 1e9))
   172→   172→  const tmp = useMemo(
   173→   173→    () => ({
   174→   174→      a: new THREE.Vector3(),
   175→   175→      b: new THREE.Vector3(),
   176→   176→      ctrl: new THREE.Vector3(),
   177→   177→      mid: new THREE.Vector3(),
   178→   178→      dir: new THREE.Vector3(),
   179→   179→      perp: new THREE.Vector3(),
   180→   180→    }),
   181→   181→    []
   182→   182→  )
   183→   183→
   184→   184→  useFrame((state) => {
   185→   185→    const m = mesh.current
   186→   186→    if (!m) return
   187→   187→    const parentPort = anchors.current?.get(parentId)?.get(parentPortId)
   188→   188→    const childPort = anchors.current?.get(childId)?.get(childPortId)
   189→   189→    if (!parentPort || !childPort) {
   190→   190→      m.visible = false
   191→   191→      return
   192→   192→    }
   193→   193→    m.visible = true
   194→   194→    ;(side === 'red' ? parentPort.red : parentPort.black).getWorldPosition(tmp.a)
   195→   195→    ;(side === 'red' ? childPort.red : childPort.black).getWorldPosition(tmp.b)
   196→   196→
   197→   197→    // Provider-side terminal fan-out.
   198→   198→    tmp.a.setX(tmp.a.x + spread)
   199→   199→    // Consumer-side perpendicular ribbon offset (only when nonzero).
   200→   200→    if (lateral !== 0) {
   201→   201→      tmp.dir.subVectors(tmp.b, tmp.a).setY(0)
   202→   202→      if (tmp.dir.lengthSq() > 1e-6) {
   203→   203→        tmp.perp.set(-tmp.dir.z, 0, tmp.dir.x).normalize().multiplyScalar(lateral)
   204→   204→        tmp.b.add(tmp.perp)
   205→   205→      }
   206→   206→    }
   207→   207→
   208→   208→    if (
   209→   209→      tmp.a.distanceToSquared(lastA.current) < 1e-8 &&
   210→   210→      tmp.b.distanceToSquared(lastB.current) < 1e-8 &&
   211→   211→      m.geometry.type === 'TubeGeometry'
   212→   212→    ) {
   213→   213→      return
   214→   214→    }
   215→   215→    lastA.current.copy(tmp.a)
   216→   216→    lastB.current.copy(tmp.b)
   217→   217→
   218→   218→    // The cable sags between the two clip points — and bows toward the
   219→   219→    // camera side, so it arcs AROUND the provider's own body instead of
   220→   220→    // disappearing behind it when the consumer sits around the back.
   221→   221→    tmp.mid.addVectors(tmp.a, tmp.b).multiplyScalar(0.5)
   222→   222→    const sagY = Math.max(0.32, Math.min(tmp.a.y, tmp.b.y) - 0.5)
   223→   223→    let bowX = 0
   224→   224→    let bowZ = 0
   225→   225→    tmp.dir.subVectors(tmp.b, tmp.a).setY(0)
   226→   226→    if (tmp.dir.lengthSq() > 1e-6) {
   227→   227→      tmp.perp.set(-tmp.dir.z, 0, tmp.dir.x).normalize()
   228→   228→      const camX = state.camera.position.x
   229→   229→      const camZ = state.camera.position.z
   230→   230→      const toCam = (camX - tmp.mid.x) * tmp.perp.x + (camZ - tmp.mid.z) * tmp.perp.z
   231→   231→      const sign = toCam >= 0 ? 1 : -1
   232→   232→      bowX = tmp.perp.x * sign * BOW
   233→   233→      bowZ = tmp.perp.z * sign * BOW
   234→   234→    }
   235→   235→    tmp.ctrl.set(tmp.mid.x + bowX, sagY, tmp.mid.z + bowZ)
   236→   236→    const curve = new THREE.QuadraticBezierCurve3(
   237→   237→      tmp.a.clone(),
   238→   238→      tmp.ctrl.clone(),
   239→   239→      tmp.b.clone()
   240→   240→    )
   241→   241→    const geom = new THREE.TubeGeometry(curve, 24, radius, 10, false)
   242→   242→    m.geometry.dispose()
   243→   243→    m.geometry = geom
   244→   244→  })
   245→   245→
   246→   246→  return (
   247→   247→    <mesh ref={mesh}>
   248→   248→      <meshStandardMaterial
   249→   249→        color={color}
   250→   250→        roughness={0.38}
   251→   251→        metalness={0.12}
   252→   252→        emissive={live ? '#b91c1c' : '#000000'}
   253→   253→        emissiveIntensity={live ? (side === 'red' ? 0.85 : 0.3) : 0}
   254→   254→      />
   255→   255→    </mesh>
   256→   256→  )
   257→   257→}

   303→   303→/** A nearby foreign component the dragged one could connect to. */
   304→   304→interface Candidate {
   305→   305→  id: string
   306→   306→  x: number
   307→   307→  z: number
   308→   308→  dist: number
   309→   309→  connectable: boolean
   310→   310→  reason: string
   311→   311→  outId: string
   312→   312→  inId: string
   313→   313→  outPortId: string
   314→   314→  inPortId: string
   315→   315→}
   316→   316→
   317→   317→/**
   318→   318→ * The nearest foreign instance within snap reach of the dragged group —
   319→   319→ * null when nothing is close enough. `connectable` + `reason` come from
   320→   320→ * the REAL-LIFE port matching: one side must provide power the other
   321→   321→ * accepts (a 9V battery onto a servo is refused — 9 V is outside the
   322→   322→ * MG996R's 4.8–7.2 V range).
   323→   323→ */
   324→   324→function findCandidate(
   325→   325→  insts: PlayInstance[],
   326→   326→  groups: React.RefObject<Map<string, THREE.Group>>,
   327→   327→  grabbedId: string,
   328→   328→  members: string[],
   329→   329→  pos: { x: number; z: number }
   330→   330→): Candidate | null {
   331→   331→  const byId = new Map(insts.map((i) => [i.id, i]))
   332→   332→  const grabbedInst = byId.get(grabbedId)
   333→   333→  if (!grabbedInst) return null
   334→   334→  let nearest: { id: string; x: number; z: number; dist: number } | null = null
   335→   335→  for (const other of insts) {
   336→   336→    if (members.includes(other.id)) continue
   337→   337→    const g = groups.current?.get(other.id)
   338→   338→    if (!g) continue
   339→   339→    const dist = Math.hypot(g.position.x - pos.x, g.position.z - pos.z)
   340→   340→    if (!nearest || dist < nearest.dist) {
   341→   341→      nearest = { id: other.id, x: g.position.x, z: g.position.z, dist }
   342→   342→    }
   343→   343→  }
   344→   344→  if (!nearest) return null
   345→   345→  const reach = footprintFor(grabbedInst.kind) + footprintFor(byId.get(nearest.id)?.kind ?? 'gltf') + SNAP_REACH
   346→   346→  if (nearest.dist >= reach) return null
   347→   347→  const targetInst = byId.get(nearest.id)
   348→   348→  if (!targetInst) return null
   349→   349→  const match = connectBetween(grabbedInst, targetInst)
   350→   350→  return {
   351→   351→    id: nearest.id,
   352→   352→    x: nearest.x,
   353→   353→    z: nearest.z,
   354→   354→    dist: nearest.dist,
   355→   355→    connectable: match.ok,
   356→   356→    reason: match.reason,
   357→   357→    outId: match.outId,
   358→   358→    inId: match.inId,
   359→   359→    outPortId: match.outPortId,
   360→   360→    inPortId: match.inPortId,
   361→   361→  }
   362→   362→}

   380→   380→        ref={mat}
   381→   381→        color={connectable ? '#10b981' : '#dc2626'}

   393→   393→ * Register the wire-clip anchors for one instance — the REAL connectors:
   394→   394→ * the battery's two brass snap terminals, the controller's VIN jack and
   395→   395→ * 5 V header, the servo's cable exit, nominal lead posts on uploads.

   397→   397→function usePortAnchors(
   398→   398→  ref: React.RefObject<THREE.Group | null>,
   399→   399→  inst: PlayInstance,
   400→   400→  anchors: React.RefObject<AnchorMap>
   401→   401→) {
   402→   402→  useEffect(() => {
   403→   403→    const g = ref.current
   404→   404→    if (!g) return
   405→   405→    const map = new Map<string, PortAnchor>()
   406→   406→    const created: THREE.Object3D[] = []
   407→   407→
   408→   408→    const addSingle = (portId: string, pos: [number, number, number]) => {
   409→   409→      const o = new THREE.Object3D()
   410→   410→      o.position.set(pos[0], pos[1], pos[2])
   411→   411→      g.add(o)
   412→   412→      created.push(o)
   413→   413→      map.set(portId, { red: o, black: o })
   414→   414→    }
   415→   415→    const addPair = (
   416→   416→      portId: string,
   417→   417→      plus: [number, number, number],
   418→   418→      minus: [number, number, number]
   419→   419→    ) => {
   420→   420→      const r = new THREE.Object3D()
   421→   421→      r.position.set(plus[0], plus[1], plus[2])
   422→   422→      const b = new THREE.Object3D()
   423→   423→      b.position.set(minus[0], minus[1], minus[2])
   424→   424→      g.add(r)
   425→   425→      g.add(b)
   426→   426→      created.push(r, b)
   427→   427→      map.set(portId, { red: r, black: b })
   428→   428→    }
   429→   429→
   430→   430→    if (inst.kind === 'battery') {
   431→   431→      // The two real PP3 snap terminals on top — red goes to + (plus).
   432→   432→      addPair(
   433→   433→        'pp3',
   434→   434→        [BATTERY_TERMINALS.plus[0], BATTERY_TERMINALS.plus[1], 0],
   435→   435→        [BATTERY_TERMINALS.minus[0], BATTERY_TERMINALS.minus[1], 0]
   436→   436→      )
   437→   437→    } else if (inst.kind === 'controller') {
   438→   438→      // VIN jack (pair) + 5 V pin header (servo ribbon).
   439→   439→      addPair('vin', ARDUINO_ANCHORS.vin.plus, ARDUINO_ANCHORS.vin.minus)
   440→   440→      addSingle('hdr', ARDUINO_ANCHORS.header)
   441→   441→    } else if (inst.kind === 'servo') {
   442→   442→      // The cable exit grommet on the rear of the body.
   443→   443→      addSingle('plug', [-2.5, 1.6, 0])
   444→   444→    } else {
   445→   445→      // Uploaded models: nominal power lead posts on top.
   446→   446→      const y = 2.35
   447→   447→      addPair('leads', [0.45, y, 0], [-0.45, y, 0])
   448→   448→    }
   449→   449→
   450→   450→    anchors.current?.set(inst.id, map)
   451→   451→    return () => {
   452→   452→      anchors.current?.delete(inst.id)
   453→   453→      for (const o of created) g.remove(o)
   454→   454→    }
   455→   455→  }, [inst.id, inst.kind, ref, anchors])
   456→   456→}
   457→   457→
   458→   458→function Instance({
   459→   459→  inst,
   460→   460→  running,
   461→   461→  powered,
   462→   462→  energized,
   463→   463→  selected,
   464→   464→  candidate,
   465→   465→  groups,
   466→   466→  anchors,
   467→   467→  onDragStart,
   468→   468→}: {
   469→   469→  inst: PlayInstance
   470→   470→  running: boolean
   471→   471→  powered: boolean
   472→   472→  energized: boolean
   473→   473→  selected: boolean
   474→   474→  candidate: LiveCandidate | null
   475→   475→  groups: React.RefObject<Map<string, THREE.Group>>
   476→   476→  anchors: React.RefObject<AnchorMap>
   477→   477→  onDragStart: (e: ThreeEvent<PointerEvent>, id: string) => void
   478→   478→}) {
   479→   479→  const ref = useRef<THREE.Group>(null)
   480→   480→
   481→   481→  useEffect(() => {
   482→   482→    const g = ref.current
   483→   483→    if (!g) return
   484→   484→    g.position.set(inst.x, 0, inst.z)
   485→   485→    groups.current?.set(inst.id, g)
   486→   486→    // Position is set imperatively; inst.x/z only matter at mount.
   487→   487→  }, [inst.id, inst.x, inst.z, groups])
   488→   488→
   489→   489→  usePortAnchors(ref, inst, anchors)
   490→   490→
   491→   491→  const live = running && powered
   492→   492→  const ringRadius = footprintFor(inst.kind)
   493→   493→
   494→   494→  return (
   495→   495→    <group ref={ref} onPointerDown={(e) => onDragStart(e, inst.id)}>
   496→   496→      {selected && (
   497→   497→        <mesh rotation-x={-Math.PI / 2} position={[0, 0.03, 0]}>
   498→   498→          <ringGeometry args={[ringRadius, ringRadius + 0.28, 64]} />
   499→   499→          <meshBasicMaterial color="#f59e0b" transparent opacity={0.9} depthWrite={false} />
   500→   500→        </mesh>
   501→   501→      )}
   502→   502→
   503→   503→      {candidate?.id === inst.id && (
   504→   504→        <PulseRing radius={ringRadius} connectable={candidate.connectable} />
   505→   505→      )}
   506→   506→
   507→   507→      {inst.kind === 'servo' ? (
   508→   508→        <ServoMotor
   509→   509→          position={[0, 0.03, 0]}
   510→   510→          interactive
   511→   511→          showCable
   512→   512→          autoSweep={live ? sweepFor(inst.id) : false}
   513→   513→        />
   514→   514→      ) : inst.kind === 'battery' ? (
   515→   515→        <Battery9V position={[0, 0.03, 0]} ledOn={running && energized} />
   516→   516→      ) : inst.kind === 'controller' ? (
   517→   517→        <ArduinoUno position={[0, 0.03, 0]} ledOn={running && powered} />
   518→   518→      ) : inst.modelUrl ? (
   519→   519→        <SpinGroup running={live}>
   520→   520→          <Suspense fallback={null}>
   521→   521→            <LoadBoundary key={inst.modelUrl}>
   522→   522→              <GltfModel key={inst.modelUrl} url={inst.modelUrl} kind={inst.kind} />
   523→   523→            </LoadBoundary>
   524→   524→          </Suspense>
   525→   525→        </SpinGroup>
   526→   526→      ) : null}
   527→   527→    </group>
   528→   528→  )
   529→   529→}

   532→   532→
   533→   533→function Scene({
   534→   534→  instances,
   535→   535→  running,
   536→   536→  poweredIds,
   537→   537→  selectedId,
   538→   538→  groups,
   539→   539→  anchors,
   540→   540→  onSelect,
   541→   541→  onDropCommit,
   542→   542→  onConnectRequest,
   543→   543→  onCandidateChange,
   544→   544→  actionsRef,
   545→   545→}: {
   546→   546→  instances: PlayInstance[]
   547→   547→  running: boolean
   548→   548→  poweredIds: Set<string>
   549→   549→  selectedId: string | null
   550→   550→  groups: React.RefObject<Map<string, THREE.Group>>
   551→   551→  anchors: React.RefObject<AnchorMap>
   552→   552→  onSelect: (id: string) => void
   553→   553→  onDropCommit: (moves: Move[], attach: Attach | null) => void
   554→   554→  /** Fired when a component is dropped near a connectable one — the app
   555→   555→      must ASK FOR PERMISSION; nothing is wired until it answers.
   556→   556→      The parent is the side that PROVIDES power. */
   557→   557→  onConnectRequest: (
   558→   558→    parentId: string,
   559→   559→    childId: string,
   560→   560→    parentPortId: string,
   561→   561→    childPortId: string
   562→   562→  ) => void
   563→   563→  /** Live proximity feedback (the pulsing-ring target), or null. */
   564→   564→  onCandidateChange: (candidate: LiveCandidate | null) => void
   565→   565→  actionsRef: React.RefObject<StageActions>
   566→   566→}) {
   567→   567→  // Read controls through the store getter (mutating a subscribed hook
   568→   568→  // value directly trips react-hooks/immutability — ServoMotor does the same).
   569→   569→  const get = useThree((s) => s.get)
   570→   570→  // Latest instances for the drag/drop logic without re-binding listeners.
   571→   571→  const instancesRef = useRef(instances)
   572→   572→  useEffect(() => {
   573→   573→    instancesRef.current = instances
   574→   574→  }, [instances])
   575→   575→
   576→   576→  // A battery's LED means "power is flowing" — lit only while it is
   577→   577→  // actually wired to something it can power.
   578→   578→  const loadedBatteries = useMemo(() => batteriesWithLoad(instances), [instances])
   579→   579→
   580→   580→  // Per-parent fan-out indices: when one battery feeds several
   581→   581→  // consumers, their wire pairs attach side by side on the terminals.
   582→   582→  const terminalSpread = useMemo(() => {
   583→   583→    const byParent = new Map<string, string[]>()
   584→   584→    for (const inst of instances) {
   585→   585→      if (!inst.parentId) continue
   586→   586→      const list = byParent.get(inst.parentId) ?? []
   587→   587→      list.push(inst.id)
   588→   588→      byParent.set(inst.parentId, list)
   589→   589→    }
   590→   590→    const spread = new Map<string, number>()
   591→   591→    for (const siblings of byParent.values()) {
   592→   592→      siblings.forEach((childId, i) => {
   593→   593→        spread.set(childId, (i - (siblings.length - 1) / 2) * TERMINAL_SPREAD)
   594→   594→      })
   595→   595→    }
   596→   596→    return spread
   597→   597→  }, [instances])
   598→   598→
   599→   599→  // Which neighbour is currently in reach (for the pulsing ring + hint).
   600→   600→  // State only for rendering; the ref keeps the last value so the drag
   601→   601→  // handler can compare without stale-closure surprises.
   602→   602→  const candidateRef = useRef<LiveCandidate | null>(null)
   603→   603→  const [candidate, setCandidate] = React.useState<LiveCandidate | null>(null)
   604→   604→  const setCandidateLive = useCallback(
   605→   605→    (c: LiveCandidate | null) => {
   606→   606→      const prev = candidateRef.current
   607→   607→      const same =
   608→   608→        (prev?.id ?? null) === (c?.id ?? null) &&
   609→   609→        (prev?.connectable ?? null) === (c?.connectable ?? null)
   610→   610→      if (same) return
   611→   611→      candidateRef.current = c
   612→   612→      setCandidate(c)
   613→   613→      onCandidateChange(c)
   614→   614→    },
   615→   615→    [onCandidateChange]
   616→   616→  )
   617→   617→
   618→   618→  const drag = useRef<{
   619→   619→    id: string
   620→   620→    members: string[]
   621→   621→    starts: Map<string, { x: number; z: number }>
   622→   622→    grabbedStart: { x: number; z: number }
   623→   623→    offX: number
   624→   624→    offZ: number
   625→   625→    moved: number
   626→   626→  } | null>(null)
   627→   627→
   628→   628→  const onDragStart = useCallback(
   629→   629→    (e: ThreeEvent<PointerEvent>, id: string) => {
   630→   630→      if (e.pointerType === 'mouse' && e.button !== 0) return
   631→   631→      e.stopPropagation()
   632→   632→      const grabbed = groups.current?.get(id)
   633→   633→      if (!grabbed) return
   634→   634→      // The whole wired group travels together.
   635→   635→      const members = [...groupMembers(instancesRef.current, id)]
   636→   636→      const starts = new Map<string, { x: number; z: number }>()
   637→   637→      for (const memberId of members) {
   638→   638→        const g = groups.current?.get(memberId)
   639→   639→        if (g) starts.set(memberId, { x: g.position.x, z: g.position.z })
   640→   640→      }
   641→   641→      drag.current = {
   642→   642→        id,
   643→   643→        members,
   644→   644→        starts,
   645→   645→        grabbedStart: { x: grabbed.position.x, z: grabbed.position.z },
   646→   646→        offX: e.point.x - grabbed.position.x,
   647→   647→        offZ: e.point.z - grabbed.position.z,
   648→   648→        moved: 0,
   649→   649→      }
   650→   650→      const controls = (get().controls ?? null) as { enabled: boolean } | null
   651→   651→      if (controls) controls.enabled = false
   652→   652→    },
   653→   653→    [get, groups]
   654→   654→  )
   655→   655→
   656→   656→  // Release anywhere (even outside the canvas) ends a drag cleanly —
   657→   657→  // committing positions and, when the piece was dropped near a
   658→   658→  // CONNECTABLE neighbour, asking for permission instead of attaching.
   659→   659→  useEffect(() => {
   660→   660→    const finish = () => {
   661→   661→      const d = drag.current
   662→   662→      drag.current = null
   663→   663→      const controls = (get().controls ?? null) as { enabled: boolean } | null
   664→   664→      if (controls) controls.enabled = true
   665→   665→      setCandidateLive(null)
   666→   666→      if (!d) return
   667→   667→
   668→   668→      if (d.moved < 0.12) {
   669→   669→        onSelect(d.id)
   670→   670→        return
   671→   671→      }
   672→   672→
   673→   673→      const insts = instancesRef.current
   674→   674→      const byId = new Map(insts.map((i) => [i.id, i]))
   675→   675→      const grabbedInst = byId.get(d.id)
   676→   676→      const grabbedGroup = groups.current?.get(d.id)
   677→   677→      if (!grabbedInst || !grabbedGroup) return
   678→   678→
   679→   679→      // Collect the group's end positions.
   680→   680→      const moves: Move[] = d.members.map((memberId) => {
   681→   681→        const g = groups.current?.get(memberId)
   682→   682→        return { id: memberId, x: g?.position.x ?? 0, z: g?.position.z ?? 0 }
   683→   683→      })
   684→   684→
   685→   685→      const cand = findCandidate(insts, groups, d.id, d.members, {
   686→   686→        x: grabbedGroup.position.x,
   687→   687→        z: grabbedGroup.position.z,
   688→   688→      })
   689→   689→
   690→   690→      if (cand && cand.connectable) {
   691→   691→        const targetInst = byId.get(cand.id)
   692→   692→        if (targetInst) {
   693→   693→          // Preview: slide the grabbed piece AND everything wired below
   694→   694→          // it into the connected spot while the user decides. The cables
   695→   695→          // only appear if the answer is "Connect".
   696→   696→          let dx = grabbedGroup.position.x - cand.x
   697→   697→          let dz = grabbedGroup.position.z - cand.z
   698→   698→          const len = Math.hypot(dx, dz)
   699→   699→          if (len < 0.001) {
   700→   700→            dx = 1
   701→   701→            dz = 0
   702→   702→          } else {
   703→   703→            dx /= len
   704→   704→            dz /= len
   705→   705→          }
   706→   706→          const gap = footprintFor(grabbedInst.kind) + footprintFor(targetInst.kind) + SNAP_GAP
   707→   707→          const snappedX = THREE.MathUtils.clamp(cand.x + dx * gap, -BOUND, BOUND)
   708→   708→          const snappedZ = THREE.MathUtils.clamp(cand.z + dz * gap, -BOUND, BOUND)
   709→   709→          const deltaX = snappedX - grabbedGroup.position.x
   710→   710→          const deltaZ = snappedZ - grabbedGroup.position.z
   711→   711→          for (const memberId of subtreeIds(insts, d.id)) {
   712→   712→            const g = groups.current?.get(memberId)
   713→   713→            if (!g) continue
   714→   714→            g.position.x = THREE.MathUtils.clamp(g.position.x + deltaX, -BOUND, BOUND)
   715→   715→            g.position.z = THREE.MathUtils.clamp(g.position.z + deltaZ, -BOUND, BOUND)
   716→   716→            const move = moves.find((m) => m.id === memberId)
   717→   717→            if (move) {
   718→   718→              move.x = g.position.x
   719→   719→              move.z = g.position.z
   720→   720→            } else {
   721→   721→              moves.push({ id: memberId, x: g.position.x, z: g.position.z })
   722→   722→            }
   723→   723→          }
   724→   724→          onDropCommit(moves, null)
   725→   725→          // The parent is whichever side PROVIDES power.
   726→   726→          onConnectRequest(cand.outId, cand.inId, cand.outPortId, cand.inPortId)
   727→   727→          return
   728→   728→        }
   729→   729→      }
   730→   730→
   731→   731→      // Nothing connectable nearby (or an impossible-in-real-life pair) —
   732→   732→      // a plain drop.
   733→   733→      onDropCommit(moves, null)
   734→   734→    }
   735→   735→    window.addEventListener('pointerup', finish)
   736→   736→    return () => window.removeEventListener('pointerup', finish)
   737→   737→  }, [get, groups, onDropCommit, onSelect, onConnectRequest, setCandidateLive])
   738→   738→
   739→   739→  // "Don't connect" — push the declined piece (and its subtree) back out
   740→   740→  // of the preview spot so it clearly isn't wired to anything, then
   741→   741→  // commit the new positions.
   742→   742→  const separate = useCallback(
   743→   743→    (childId: string, parentId: string) => {
   744→   744→      const insts = instancesRef.current
   745→   745→      const child = insts.find((i) => i.id === childId)
   746→   746→      const parent = insts.find((i) => i.id === parentId)
   747→   747→      const childGroup = groups.current?.get(childId)
   748→   748→      const parentGroup = groups.current?.get(parentId)
   749→   749→      if (!child || !parent || !childGroup || !parentGroup) return
   750→   750→
   751→   751→      let dx = childGroup.position.x - parentGroup.position.x
   752→   752→      let dz = childGroup.position.z - parentGroup.position.z
   753→   753→      const len = Math.hypot(dx, dz)
   754→   754→      if (len < 0.001) {
   755→   755→        dx = 1
   756→   756→        dz = 0
   757→   757→      } else {
   758→   758→        dx /= len
   759→   759→        dz /= len
   760→   760→      }
   761→   761→      const clearGap = footprintFor(child.kind) + footprintFor(parent.kind) + DECLINE_GAP
   762→   762→      const targetX = THREE.MathUtils.clamp(parentGroup.position.x + dx * clearGap, -BOUND, BOUND)
   763→   763→      const targetZ = THREE.MathUtils.clamp(parentGroup.position.z + dz * clearGap, -BOUND, BOUND)
   764→   764→      const deltaX = targetX - childGroup.position.x
   765→   765→      const deltaZ = targetZ - childGroup.position.z
   766→   766→
   767→   767→      const moves: Move[] = []
   768→   768→      for (const memberId of subtreeIds(insts, childId)) {
   769→   769→        const g = groups.current?.get(memberId)
   770→   770→        if (!g) continue
   771→   771→        g.position.x = THREE.MathUtils.clamp(g.position.x + deltaX, -BOUND, BOUND)
   772→   772→        g.position.z = THREE.MathUtils.clamp(g.position.z + deltaZ, -BOUND, BOUND)
   773→   773→        moves.push({ id: memberId, x: g.position.x, z: g.position.z })
   774→   774→      }
   775→   775→      onDropCommit(moves, null)
   776→   776→    },
   777→   777→    [groups, onDropCommit]
   778→   778→  )
   779→   779→
   780→   780→  // Expose the imperative actions to the React layer (decline handling).
   781→   781→  useEffect(() => {
   782→   782→    if (actionsRef.current) actionsRef.current.separate = separate
   783→   783→  }, [actionsRef, separate])
   784→   784→
   785→   785→  const handleMove = (e: ThreeEvent<PointerEvent>) => {
   786→   786→    const d = drag.current
   787→   787→    if (!d) return
   788→   788→    const grabbedGroup = groups.current?.get(d.id)
   789→   789→    if (!grabbedGroup) return
   790→   790→    const x = THREE.MathUtils.clamp(e.point.x - d.offX, -BOUND, BOUND)
   791→   791→    const z = THREE.MathUtils.clamp(e.point.z - d.offZ, -BOUND, BOUND)
   792→   792→    const deltaX = x - d.grabbedStart.x
   793→   793→    const deltaZ = z - d.grabbedStart.z
   794→   794→    for (const memberId of d.members) {
   795→   795→      const g = groups.current?.get(memberId)
   796→   796→      const start = d.starts.get(memberId)
   797→   797→      if (!g || !start) continue
   798→   798→      g.position.x = THREE.MathUtils.clamp(start.x + deltaX, -BOUND, BOUND)
   799→   799→      g.position.z = THREE.MathUtils.clamp(start.z + deltaZ, -BOUND, BOUND)
   800→   800→    }
   801→   801→    d.moved = Math.max(d.moved, Math.hypot(x - d.grabbedStart.x, z - d.grabbedStart.z))
   802→   802→
   803→   803→    // Live proximity: pulse the nearest neighbour when it's in reach.
   804→   804→    const cand = findCandidate(instancesRef.current, groups, d.id, d.members, {
   805→   805→      x: grabbedGroup.position.x,
   806→   806→      z: grabbedGroup.position.z,
   807→   807→    })
   808→   808→    setCandidateLive(
   809→   809→      cand ? { id: cand.id, connectable: cand.connectable, reason: cand.reason } : null
   810→   810→    )
   811→   811→  }
   812→   812→
   813→   813→  return (
   814→   814→    <>
   815→   815→      <ambientLight intensity={0.42} />
   816→   816→      <directionalLight
   817→   817→        position={[5, 9, 6]}
   818→   818→        intensity={2.3}
   819→   819→        castShadow
   820→   820→        shadow-mapSize={[2048, 2048]}
   821→   821→        shadow-camera-left={-18}
   822→   822→        shadow-camera-right={18}
   823→   823→        shadow-camera-top={18}
   824→   824→        shadow-camera-bottom={-18}
   825→   825→        shadow-camera-near={1}
   826→   826→        shadow-camera-far={40}
   827→   827→        shadow-bias={-0.0003}
   828→   828→        shadow-normalBias={0.02}
   829→   829→      />
   830→   830→      <directionalLight position={[-7, 4, -6]} intensity={0.55} />
   831→   831→
   832→   832→      <Environment resolution={256} frames={1} environmentIntensity={0.62}>
   833→   833→        <Lightformer form="rect" intensity={3} position={[0, 6, 0]} rotation-x={Math.PI / 2} scale={[9, 9, 1]} color="#ffffff" />
   834→   834→        <Lightformer form="rect" intensity={1.8} position={[6, 3, 4]} rotation-y={-Math.PI / 3} scale={[6, 3, 1]} color="#fff2df" />
   835→   835→        <Lightformer form="rect" intensity={1.8} position={[-6, 2, -3]} rotation-y={Math.PI / 2.5} scale={[5, 3, 1]} color="#eef2f8" />
   836→   836→      </Environment>
   837→   837→
   838→   838→      <Grid
   839→   839→        position={[0, 0.001, 0]}
   840→   840→        args={[90, 90]}
   841→   841→        cellSize={0.55}
   842→   842→        cellThickness={0.6}
   843→   843→        cellColor="#d3d6dd"
   844→   844→        sectionSize={2.75}
   845→   845→        sectionThickness={1}
   846→   846→        sectionColor="#b9bfc9"
   847→   847→        fadeDistance={40}
   848→   848→        fadeStrength={1.4}
   849→   849→        infiniteGrid
   850→   850→        followCamera={false}
   851→   851→      />
   852→   852→      <ContactShadows
   853→   853→        position={[0, 0.02, 0]}
   854→   854→        opacity={0.5}
   855→   855→        scale={44}
   856→   856→        blur={2.4}
   857→   857→        far={4}
   858→   858→        resolution={512}
   859→   859→        color="#18181b"
   860→   860→      />
   861→   861→
   862→   862→      {/* Invisible ground plane — the drag surface */}
   863→   863→      <mesh rotation-x={-Math.PI / 2} onPointerMove={handleMove}>
   864→   864→        <planeGeometry args={[200, 200]} />
   865→   865→        <meshBasicMaterial transparent opacity={0} depthWrite={false} />
   866→   866→      </mesh>
   867→   867→
   868→   868→      {instances.map((inst) => (
   869→   869→        <Instance
   870→   870→          key={inst.id}
   871→   871→          inst={inst}
   872→   872→          running={running}
   873→   873→          powered={poweredIds.has(inst.id)}
   874→   874→          energized={inst.kind === 'battery' && loadedBatteries.has(inst.id)}
   875→   875→          selected={inst.id === selectedId}
   876→   876→          candidate={candidate}
   877→   877→          groups={groups}
   878→   878→          anchors={anchors}
   879→   879→          onDragStart={onDragStart}
   880→   880→        />
   881→   881→      ))}
   882→   882→
   883→   883→      {/* Cables for every wired link — between the REAL mated ports */}
   884→   884→      {instances
   885→   885→        .filter((i) => i.parentId)
   886→   886→        .map((i) => {
   887→   887→          const inPort = getPort(i.kind, i.childPort) ?? ELEC_PORTS[i.kind]?.[0]
   888→   888→          return (
   889→   889→            <LinkWire
   890→   890→              key={`wire-${i.id}`}
   891→   891→              anchors={anchors}
   892→   892→              parentId={i.parentId as string}
