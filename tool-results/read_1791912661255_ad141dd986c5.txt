     1→/**
     2→ * The component schema — the platform's single source of truth.
     3→ *
     4→ * This file is what turns "a nice 3D viewer" into Pillar 1 of the robotics
     5→ * creation platform. Every library component is described as DATA:
     6→ *
     7→ *   identity    → what it is, who makes it, what it costs (real-world commerce)
     8→ *   specs       → the headline engineering numbers from the datasheet
     9→ *   physics     → mass, materials, electrical load — what a simulation engine
    10→ *                 (Godot later) needs to make the virtual behave like the real
    11→ *   ai          → a machine-readable profile the AI generation layer (Pillar 2)
    12→ *                 reads when a user says "build me a robot that avoids walls"
    13→ *   model       → where the 3D geometry comes from (procedural today, glTF
    14→ *                 from OpenSCAD/STL exports later — the schema accepts both)
    15→ *
    16→ * The geometry itself stays procedural in TypeScript (see ServoMotor.tsx):
    17→ * procedural models can be parameterized, are tiny to download, and render
    18→ * interactively. The DB stores the catalog + ownership; this file is the
    19→ * definition. Everything here is JSON-serializable on purpose so it can be
    20→ * served straight to browsers, the AI layer, and the future build-instruction
    21→ * exporter without transformation.
    22→ */
    23→
    24→import type { ComponentModel } from './connections'
    25→import { SERVO_MODEL } from './connections'
    26→
    27→/* ------------------------------------------------------------------ */
    28→/* Types                                                               */
    29→/* ------------------------------------------------------------------ */
    30→
    31→export type ComponentStatus = 'available' | 'soon'
    32→export type ComponentCategory = 'Actuators' | 'Sensors' | 'Power' | 'Controller'
    33→
    34→export interface SpecRow {
    35→  label: string
    36→  value: string
    37→}
    38→
    39→export interface PhysicsProperties {
    40→  /** Mass in grams — the number a physics engine steps with. */
    41→  mass: number
    42→  /** Footprint in mm: length × width × height. */
    43→  dimensions: { l: number; w: number; h: number }
    44→  /** What each part is made of — feeds material physics + cost realism. */
    45→  materials: Array<{ part: string; material: string }>
    46→  /** Electrical load profile. */
    47→  electrical: Array<{ label: string; value: string }>
    48→  /** Mechanical ratings relevant to simulation. */
    49→  mechanical: Array<{ label: string; value: string }>
    50→  /** One-paragraph note on how the sim engine should treat this part. */
    51→  simNotes: string
    52→}
    53→
    54→export interface CommerceInfo {
    55→  manufacturer: string
    56→  partNumber: string
    57→  /** Typical street price range in USD. */
    58→  priceUsd: [number, number]
    59→  /** INR equivalent — the first market this platform targets. */
    60→  priceInr: [number, number]
    61→  availability: string
    62→  /** Where a student can actually buy it. */
    63→  vendors: string[]
    64→}
    65→
    66→export interface AiProfile {
    67→  /** One line the AI layer reads first. */
    68→  summary: string
    69→  /** What this component can do in an assembly. */
    70→  capabilities: string[]
    71→  /** Hard limits the AI must respect when designing. */
    72→  constraints: string[]
    73→  /** Situations where this is the right pick. */
    74→  bestFor: string[]
    75→}
    76→
    77→export type ModelSource =
    78→  | {
    79→      /** Code-generated parametric geometry (an R3F component export). */
    80→      kind: 'procedural'
    81→      /** Name of the exported React Three Fiber component. */
    82→      component: string
    83→      /** True if the mesh matches the spec-sheet dimensions exactly. */
    84→      dimensionAccurate: boolean
    85→      note: string
    86→    }
    87→  | {
    88→      /** External asset (future: OpenSCAD → STL → glTF pipeline). */
    89→      kind: 'gltf'
    90→      url: string
    91→    }
    92→
    93→export interface RoboticsComponent {
    94→  id: string
    95→  name: string
    96→  category: ComponentCategory
    97→  status: ComponentStatus
    98→  tagline: string
    99→  /** Long-form description for the component detail page. */
   100→  description: string
   101→  specs: SpecRow[]
   102→  physics: PhysicsProperties
   103→  commerce: CommerceInfo
   104→  ai: AiProfile
   105→  model: ModelSource
   106→  /** Connection contract (ports + DOF) — defined in connections.ts. */
   107→  connectionModel?: ComponentModel
   108→}

   613→/**
   614→ * The machine-readable catalog — exactly what the AI generation layer
   615→ * (Pillar 2) will be fed as tool/context when a user says "build me a
   616→ * line-following robot that avoids obstacles". Geometry references are
   617→ * stripped; capabilities, constraints, ports, physics and cost remain.
   618→ */
   619→export function aiCatalogPayload() {
   620→  return {
   621→    schema: 'robotics-platform.catalog/1.0',
   622→    generatedAt: new Date().toISOString(),
   623→    componentCount: COMPONENTS.length,
   624→    components: COMPONENTS.map((c) => ({
   625→      id: c.id,
   626→      name: c.name,
   627→      category: c.category,
   628→      status: c.status,
   629→      tagline: c.tagline,
   630→      mass_g: c.physics.mass,
   631→      dimensions_mm: c.physics.dimensions,
   632→      price_usd: c.commerce.priceUsd,
   633→      part_number: c.commerce.partNumber,
   634→      ports: c.connectionModel?.ports.length ?? 0,
   635→      dof: c.connectionModel?.dof.map((d) => ({
   636→        id: d.id,
   637→        kind: d.kind,
   638→        range_deg: d.range,
   639→        axis: d.axis,
   640→      })) ?? [],
   641→      ai: c.ai,
   642→      key_specs: c.specs.slice(0, 4).map((s) => `${s.label}: ${s.value}`),
   643→    })),
   644→  }
   645→}
